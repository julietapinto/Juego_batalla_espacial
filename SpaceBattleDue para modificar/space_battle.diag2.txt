.syntax unified
.cpu cortex-m3
.thumb
.equ PMC,0x400e0600
.equ PIOA,0x400e0e00
.equ PIOB,0x400e1000
.equ PIOC,0x400e1200
.equ ADC,0x400c0000
.equ TWI,0x40090000
.equ TC,0x40080000
.equ OLED_ADDR,0x3c
.equ UART,0x400e0800
.macro WRITE addr,val
 ldr r0,=\addr
 ldr r1,=\val
 str r1,[r0]
.endm
.section .vectors,"a",%progbits
.global vectors
vectors:
.word _stack,Reset_Handler
.rept 14
.word Default_Handler
.endr
.rept 27
.word Default_Handler
.endr
.word TC0_Handler
.rept 17
.word Default_Handler
.endr
.section .text
.thumb_func
.global Reset_Handler
Reset_Handler:
 cpsid i
 WRITE 0xe000ed08,vectors
 ldr r0,=_sbss
 ldr r1,=_ebss
 movs r2,#0
1: cmp r0,r1
 itt lo
 strlo r2,[r0],#4
 blo 1b
 WRITE 0x400e1a54,0x8000 // watchdog disable
 // Five flash wait states before raising master clock.
 WRITE 0x400e0a00,0x500
 WRITE 0x400e0c00,0x500
 WRITE PMC+0x20,0x00370809
 ldr r0,=PMC
1: ldr r1,[r0,#0x68]
 tst r1,#1
 beq 1b
 WRITE PMC+0x20,0x01370809
 ldr r0,=PMC
1: ldr r1,[r0,#0x68]
 tst r1,#0x10000
 beq 1b
 WRITE PMC+0x28,0x200d3f01 // 12 MHz *14 =168 MHz
 ldr r0,=PMC
1: ldr r1,[r0,#0x68]
 tst r1,#2
 beq 1b
 WRITE PMC+0x30,0x11 // MAIN /2, then PLLA /2
 bl mck_wait
 WRITE PMC+0x30,0x12
 bl mck_wait
 WRITE PMC+0x10,0x08803900 // PIO A/B/C, TWI1, TC0
 WRITE PMC+0x100,0x20 // ADC ID37
 // UART hacia Programming Port, PA8/PA9, 115200 8N1.
 ldr r0,=PIOA
 ldr r1,[r0,#0x70]
 bic r1,r1,#0x300
 str r1,[r0,#0x70]
 WRITE PIOA+4,0x300
 WRITE UART,0xac
 WRITE UART+4,0x800
 WRITE UART+0x20,46
 WRITE UART,0x50
 WRITE PIOB,0x08000000
 WRITE PIOB+0x10,0x08000000
 WRITE PIOB+0x34,0x08000000
 WRITE PIOA,0x00800000
 WRITE PIOA+0x14,0x00800000
 WRITE PIOA+0x64,0x00800000
 WRITE PIOC,0x00800000
 WRITE PIOC+0x14,0x00800000
 WRITE PIOC+0x64,0x00800000
 WRITE PIOA+0x04,0x01010000 // analog A0/A1
 WRITE PIOA+0x60,0x01010000
 // Peripheral A on PB12/PB13.
 ldr r0,=PIOB
 ldr r1,[r0,#0x70]
 bic r1,r1,#0x3000
 str r1,[r0,#0x70]
 WRITE PIOB+4,0x3000
 WRITE ADC,1
 WRITE ADC+4,0x13380100 // ADC 21 MHz, startup 512 clocks, tracking 4
 WRITE ADC+0x10,0xc0
 WRITE TWI,0x80
 WRITE TWI,0x24
 WRITE TWI+0x10,0x6565 // 84MHz/(2*(101+4)) =400 kHz
 ldr r0,=page_cmd
 movs r1,#2
 strb r1,[r0,#1]
 movs r1,#0x10
 strb r1,[r0,#2]
 bl oled_init
 WRITE TC+4,0x0000c000 // TIMER_CLOCK1 MCK/2, waveform UP_RC
 WRITE TC+0x1c,1400000 // 42 MHz / 30
 WRITE TC+0x24,0x10
 WRITE 0xe000e280,0x08000000
 WRITE 0xe000e100,0x08000000
 WRITE TC,5
 cpsie i
main:
 // Atomic check plus WFI prevents a lost wakeup.
 cpsid i
 ldr r0,=ticks
 ldr r1,[r0]
 ldr r2,=consumed
 ldr r3,[r2]
 cmp r1,r3
 bne 2f
 dsb
 wfi
 cpsie i
 b main
2: str r1,[r2]
 cpsie i
 movs r0,#1
 bl trace_stage
 bl input
 movs r0,#2
 bl trace_stage
 bl update
 movs r0,#3
 bl trace_stage
 bl render
 movs r0,#4
 bl trace_stage
 bl oled_flush
 movs r0,#5
 bl trace_stage
 bl serial_report
 b main
.thumb_func
mck_wait:
 ldr r0,=PMC
1: ldr r1,[r0,#0x68]
 tst r1,#8
 beq 1b
 bx lr
.thumb_func
Default_Handler:
 cpsid i
 movs r0,#70
 bl uart_putc
 movs r0,#65
 bl uart_putc
 movs r0,#85
 bl uart_putc
 movs r0,#76
 bl uart_putc
 movs r0,#84
 bl uart_putc
 movs r0,#10
 bl uart_putc
 b Default_Handler
.thumb_func
TC0_Handler:
 ldr r0,=TC
 ldr r1,[r0,#0x20]
 ldr r0,=ticks
 ldr r1,[r0]
 adds r1,#1
 str r1,[r0]
 // LED L: PB27, cambiar cada 16 ticks.
 tst r1,#15
 bne irq_done
 ldr r0,=PIOB
 ldr r2,[r0,#0x38]
 tst r2,#0x08000000
 ldr r2,=0x08000000
 ite eq
 streq r2,[r0,#0x30]
 strne r2,[r0,#0x34]
irq_done:
 bx lr
.ltorg
// I2C writes, bounded waits. A failed OLED stays recoverable.
// r0 mask; returns r0=0 success,1 error.
.thumb_func
wait_twi:
 ldr r1,=TWI
 ldr r2,=100000
1: ldr r3,[r1,#0x20]
 ldr r12,=last_twi_sr
 str r3,[r12]
 tst r3,#0x100
 bne 3f
 tst r3,r0
 bne 2f
 subs r2,#1
 bne 1b
3: movs r0,#1
 bx lr
2: movs r0,#0
 bx lr
// r0 buffer, r1 length, r2 control byte. Returns success/error.
.thumb_func
twi_send:
 push {r4-r8,lr}
 mov r4,r0
 mov r5,r1
 mov r6,r2
 ldr r7,=TWI
 ldr r0,=(OLED_ADDR<<16)
 str r0,[r7,#4]
 str r6,[r7,#0x34]
1: movs r0,#4
 bl wait_twi
 cmp r0,#0
 bne 3f
 cmp r5,#0
 beq 2f
 ldrb r0,[r4],#1
 str r0,[r7,#0x34]
 subs r5,#1
 b 1b
2: movs r0,#2
 str r0,[r7]
 movs r0,#1
 bl wait_twi
 cmp r0,#0
 bne 3f
 pop {r4-r8,pc}
3: ldr r1,=i2c_errors
 ldr r2,[r1]
 adds r2,#1
 str r2,[r1]
 movs r0,#0x80
 str r0,[r7]
 movs r0,#0x24
 str r0,[r7]
 ldr r0,=0x6565
 str r0,[r7,#0x10]
 movs r0,#1
 pop {r4-r8,pc}
.thumb_func
oled_init:
 push {r4,lr}
 ldr r4,=2000000
1: subs r4,#1
 bne 1b
 ldr r0,=init_cmd
 movs r1,#25
 movs r2,#0
 bl twi_send
 pop {r4,pc}
.thumb_func
oled_flush:
 push {r4-r6,lr}
 movs r4,#0
 ldr r5,=framebuffer
1: ldr r0,=page_cmd
 orr r1,r4,#0xb0
 strb r1,[r0]
 movs r1,#3
 movs r2,#0
 bl twi_send
 cmp r0,#0
 bne 2f
 mov r0,r5
 movs r1,#128
 movs r2,#0x40
 bl twi_send
 cmp r0,#0
 bne 2f
 adds r5,#128
 adds r4,#1
 cmp r4,#8
 bne 1b
2: pop {r4-r6,pc}
.ltorg
.thumb_func
input:
 push {r4,lr}
 ldr r4,=ADC
 movs r0,#2
 str r0,[r4]
 ldr r2,=100000
1: ldr r0,[r4,#0x30]
 and r0,r0,#0xc0
 cmp r0,#0xc0
 beq 2f
 subs r2,#1
 bne 1b
 b 3f
2: ldr r0,[r4,#0x6c] // CDR7 X
 ldr r1,=ax
 str r0,[r1]
 ldr r0,[r4,#0x68] // CDR6 Y
 str r0,[r1,#4]
3: ldr r0,=PIOA
 ldr r0,[r0,#0x3c]
 tst r0,#0x800000
 ite eq
 moveq r0,#1
 movne r0,#0
 ldr r1,=PIOC
 ldr r1,[r1,#0x3c]
 tst r1,#0x800000
 it eq
 orreq r0,r0,#2
 // Two consecutive equal samples debounce both buttons.
 ldr r1,=raw_buttons
 ldr r2,[r1]
 str r0,[r1]
 cmp r0,r2
 bne 4f
 ldr r1,=buttons
 ldr r2,[r1]
 str r0,[r1]
 bic r0,r0,r2
 ldr r1,=pressed
 str r0,[r1]
 pop {r4,pc}
4: movs r0,#0
 ldr r1,=pressed
 str r0,[r1]
 pop {r4,pc}
.thumb_func
new_game:
 ldr r0,=game
 movs r1,#1
 str r1,[r0] // state
 movs r1,#60
 str r1,[r0,#4]
 movs r1,#52
 str r1,[r0,#8]
 movs r1,#0
 str r1,[r0,#12] // score
 str r1,[r0,#20] // bullet active
 str r1,[r0,#32] // frame
 movs r1,#3
 str r1,[r0,#16] // lives
 ldr r0,=enemies
 movs r1,#0
1: lsls r2,r1,#4
 adds r2,#12
 str r2,[r0],#4
 movs r2,#12
 str r2,[r0],#4
 adds r1,#1
 cmp r1,#5
 bne 1b
 bx lr
.thumb_func
update:
 push {r4-r8,lr}
 ldr r4,=game
 ldr r0,[r4]
 cmp r0,#1
 beq 1f
 ldr r0,=pressed
 ldr r0,[r0]
 cmp r0,#0
 beq 9f
 bl new_game
 b 9f
1: ldr r0,[r4,#32]
 adds r0,#1
 str r0,[r4,#32]
 // Independent X/Y dead zones. Two pixels/frame.
 ldr r1,=ax
 ldr r2,[r1]
 ldr r0,[r4,#4]
 bl axis
 cmp r0,#120
 it hi
 movhi r0,#120
 str r0,[r4,#4]
 ldr r1,=ay
 ldr r2,[r1]
 ldr r0,[r4,#8]
 bl axis
 cmp r0,#10
 it lo
 movlo r0,#10
 cmp r0,#56
 it hi
 movhi r0,#56
 str r0,[r4,#8]
 ldr r0,[r4,#20]
 cmp r0,#0
 bne 2f
 ldr r0,=buttons
 ldr r0,[r0]
 tst r0,#2
 beq 3f
 movs r0,#1
 str r0,[r4,#20]
 ldr r0,[r4,#4]
 adds r0,#3
 str r0,[r4,#24]
 ldr r0,[r4,#8]
 str r0,[r4,#28]
2: ldr r0,[r4,#28]
 subs r0,#4
 cmp r0,#9
 bge 21f
 movs r0,#0
 str r0,[r4,#20]
21: str r0,[r4,#28]
3: ldr r5,=enemies
 movs r6,#5
4: ldr r7,[r5] // enemy X
 ldr r8,[r5,#4] // Y
 ldr r0,[r4,#32]
 tst r0,#1
 it eq
 addeq r8,#1
 cmp r8,#57
 bge 7f
 ldr r0,[r4,#20]
 cmp r0,#0
 beq 5f
 ldr r0,[r4,#24]
 subs r0,r0,r7
 cmp r0,#7
 bhi 5f
 ldr r0,[r4,#28]
 subs r0,r0,r8
 adds r0,#2
 cmp r0,#9
 bhi 5f
 movs r0,#0
 str r0,[r4,#20]
 ldr r0,[r4,#12]
 adds r0,#10
 ldr r1,=9999
 cmp r0,r1
 it hi
 movhi r0,#0
 str r0,[r4,#12]
 b 8f
5: // rectangle overlap ship (8x8) and enemy (8x8)
 ldr r0,[r4,#4]
 subs r0,r0,r7
 adds r0,#7
 cmp r0,#14
 bhi 6f
 ldr r0,[r4,#8]
 subs r0,r0,r8
 adds r0,#7
 cmp r0,#14
 bhi 6f
7: ldr r0,[r4,#16]
 cmp r0,#0
 beq 8f
 subs r0,#1
 str r0,[r4,#16]
 bne 8f
 movs r0,#2
 str r0,[r4]
8: movs r8,#10
 // Deterministic changing spawn positions, no runtime library.
 ldr r0,[r4,#32]
 adds r7,r7,r0
 adds r7,#29
 movs r1,#120
 udiv r2,r7,r1
 mls r7,r2,r1,r7
6: str r7,[r5]
 str r8,[r5,#4]
 adds r5,#8
 subs r6,#1
 bne 4b
9: pop {r4-r8,pc}
.thumb_func
axis:

1: ldr r3,=2700

// Pixel OR, clips in both axes. r0=x r1=y.
.thumb_func
pixel:
 cmp r0,#128
 bhs 1f
 cmp r1,#64
 bhs 1f
 lsrs r2,r1,#3
 add r0,r0,r2,lsl #7
 and r1,r1,#7
 movs r2,#1
 lsl r2,r2,r1
 ldr r3,=framebuffer
 ldrb r1,[r3,r0]
 orrs r1,r2
 strb r1,[r3,r0]
1: bx lr
// Sprite: r0=x r1=y r2=8 columns, bit0 at top.
.thumb_func
sprite:
 push {r4-r10,lr}
 mov r4,r0
 mov r5,r1
 mov r6,r2
 movs r7,#0
1: ldrb r8,[r6,r7]
 movs r9,#0
2: tst r8,#1
 beq 3f
 add r0,r4,r7
 add r1,r5,r9

3: lsr r8,r8,#1
 adds r9,#1
 cmp r9,#8

 cmp r7,#8

// UART de diagnostico. No usa Arduino API ni pila USB.
.section .text
.thumb_func
uart_putc:
 ldr r1,=UART
 ldr r3,=100000
1: ldr r2,[r1,#0x14]
 tst r2,#2
 bne 2f
 subs r3,#1
 bne 1b
 bx lr
2: str r0,[r1,#0x1c]
 bx lr
.thumb_func
uart_string:
 push {r4,lr}
 mov r4,r0
1: ldrb r0,[r4],#1
 cmp r0,#0
 beq 2f
 bl uart_putc
 b 1b
2: pop {r4,pc}
.thumb_func
uart_hex:
 push {r4-r6,lr}
 mov r4,r0
 movs r5,#8
1: lsr r0,r4,#28
 cmp r0,#9
 ite ls
 addls r0,#48
 addhi r0,#55
 bl uart_putc
 lsl r4,r4,#4
 subs r5,#1
 bne 1b
 pop {r4-r6,pc}
// Etapas solo durante los primeros dos cuadros.
.thumb_func
trace_stage:
 push {r4,lr}
 ldr r1,=serial_frames
 ldr r1,[r1]
 cmp r1,#2
 bhs 1f
 adds r0,#48
 bl uart_putc
 movs r0,#32
 bl uart_putc
1: pop {r4,pc}
.thumb_func
serial_report:
 push {r4,lr}
 ldr r0,=serial_frames
 ldr r1,[r0]

 beq 1f
 ldr r0,=ticks
